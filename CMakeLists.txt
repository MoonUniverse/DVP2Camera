cmake_minimum_required(VERSION 3.18)
project(dvp_monitor LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(DVP_SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/DVP2-Linux64_231122/DVP2-Linux64" CACHE PATH "Unpacked vendor SDK")
find_package(PkgConfig REQUIRED)
pkg_check_modules(JSONC REQUIRED IMPORTED_TARGET json-c)
find_package(OpenCV REQUIRED COMPONENTS core imgproc highgui videoio)
find_path(DVP_INCLUDE DVPCamera.h PATHS "${DVP_SDK_ROOT}/Demo" NO_DEFAULT_PATH REQUIRED)
find_library(DVP_LIBRARY dvp PATHS "${DVP_SDK_ROOT}" NO_DEFAULT_PATH REQUIRED)
add_executable(dvp_monitor src/monitor.cpp)
target_include_directories(dvp_monitor PRIVATE "${DVP_INCLUDE}" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(dvp_monitor PRIVATE ${OpenCV_LIBS} PkgConfig::JSONC)
target_compile_options(dvp_monitor PRIVATE -Wall -Wextra -Wpedantic)
set_target_properties(dvp_monitor PROPERTIES BUILD_RPATH "${DVP_SDK_ROOT}")
enable_testing()
add_test(NAME help COMMAND dvp_monitor --help)
add_test(NAME synthetic_preview COMMAND dvp_monitor --synthetic --headless --duration 0.2)
add_test(NAME invalid_fps COMMAND dvp_monitor --synthetic --headless --fps nan)
set_tests_properties(invalid_fps PROPERTIES WILL_FAIL TRUE)

# The vendor loader searches beside /proc/self/exe for helper libraries/plugins.
file(GLOB DVP_RUNTIME_FILES "${DVP_SDK_ROOT}/*.so" "${DVP_SDK_ROOT}/*.so.*" "${DVP_SDK_ROOT}/dsdbg*.ini")
foreach(runtime_file IN LISTS DVP_RUNTIME_FILES)
    get_filename_component(runtime_name "${runtime_file}" NAME)
    file(CREATE_LINK "${runtime_file}" "${CMAKE_CURRENT_BINARY_DIR}/${runtime_name}" SYMBOLIC)
endforeach()

target_compile_definitions(dvp_monitor PRIVATE DVP_PLUGIN_DIR="${DVP_SDK_ROOT}" DVP_LIBRARY_PATH="${DVP_LIBRARY}")
target_link_libraries(dvp_monitor PRIVATE ${CMAKE_DL_LIBS})
add_test(NAME invalid_exposure COMMAND dvp_monitor --synthetic --headless --exposure-us 0)
add_test(NAME invalid_gain COMMAND dvp_monitor --synthetic --headless --gain nan)
set_tests_properties(invalid_exposure invalid_gain PROPERTIES WILL_FAIL TRUE)
add_test(NAME conflicting_auto_exposure COMMAND dvp_monitor --synthetic --auto --exposure-us 4000)
add_test(NAME conflicting_auto_gain COMMAND dvp_monitor --synthetic --auto --gain 1)
set_tests_properties(conflicting_auto_exposure conflicting_auto_gain PROPERTIES WILL_FAIL TRUE)

add_test(NAME invalid_codec COMMAND dvp_monitor --synthetic --headless --codec invalid)
set_tests_properties(invalid_codec PROPERTIES WILL_FAIL TRUE)
add_executable(timeline_test tests/timeline_test.cpp)
target_include_directories(timeline_test PRIVATE src ${OpenCV_INCLUDE_DIRS})
target_link_libraries(timeline_test PRIVATE ${OpenCV_LIBS})
add_test(NAME recording_timeline COMMAND timeline_test)
